# Write your own code
n = df.shape[0]

random_numbers = np.random.uniform(0, 1, n)

df['random_var'] = random_numbers

groupA = df.query("random_var < 0.4")

groupB = df.query("random_var >= 0.4")

print(f"Total number of players: {n}")
print(f"Shape of groupA: {groupA.shape}")
print(f"Shape of groupB: {groupB.shape}")
